```typescript
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { buildApp } from '../../src/app.js';
import type { FastifyInstance } from 'fastify';

describe('Auth Simplification Integration Tests', () => {
  let app: FastifyInstance;
  let validToken: string;

  beforeAll(async () => {
    app = await buildApp();
    await app.ready();

    // Get a valid token for authenticated tests
    const loginResponse = await app.inject({
      method: 'POST',
      url: '/auth/login',
      payload: {
        username: process.env.SYNC_USERNAME || 'admin',
        password: process.env.TEST_PASSWORD || 'test-password'
      }
    });
    const loginData = JSON.parse(loginResponse.payload);
    validToken = loginData.token;
  });

  afterAll(async () => {
    await app.close();
  });

  describe('POST /auth/login', () => {
    it('should return token with expiresIn field', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/auth/login',
        payload: {
          username: process.env.SYNC_USERNAME || 'admin',
          password: process.env.TEST_PASSWORD || 'test-password'
        }
      });

      expect(response.statusCode).toBe(200);
      const data = JSON.parse(response.payload);
      expect(data.success).toBe(true);
      expect(data.token).toBeDefined();
      expect(data.expiresIn).toBeDefined();
      expect(typeof data.expiresIn).toBe('number');
    });

    it('should return specific error for wrong username', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/auth/login',
        payload: { username: 'wrong', password: 'test' }
      });

      expect(response.statusCode).toBe(401);
      const data = JSON.parse(response.payload);
      expect(data.success).toBe(false);
      // Note: Login may use generic message for security
    });
  });

  describe('POST /auth/refresh', () => {
    it('should return new token when given valid token', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/auth/refresh',
        headers: {
          authorization: `Bearer ${validToken}`
        }
      });

      expect(response.statusCode).toBe(200);
      const data = JSON.parse(response.payload);
      expect(data.success).toBe(true);
      expect(data.token).toBeDefined();
      expect(data.expiresIn).toBeDefined();
      expect(data.tokenType).toBe('Bearer');
    });

    it('should return TOKEN_MISSING when no token provided', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/auth/refresh'
      });

      expect(response.statusCode).toBe(401);
      const data = JSON.parse(response.payload);
      expect(data.success).toBe(false);
      expect(data.error).toBe('TOKEN_MISSING');
    });

    it('should return TOKEN_INVALID for malformed token', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/auth/refresh',
        headers: {
          authorization: 'Bearer invalid-token-format'
        }
      });

      expect(response.statusCode).toBe(401);
      const data = JSON.parse(response.payload);
      expect(data.success).toBe(false);
      expect(data.error).toBe('TOKEN_INVALID');
    });
  });

  describe('GET /api/auth/diagnostics', () => {
    it('should return token metadata for authenticated user', async () => {
      const response = await app.inject({
        method: 'GET',
        url: '/api/auth/diagnostics',
        headers: {
          authorization: `Bearer ${validToken}`
        }
      });

      expect(response.statusCode).toBe(200);
      const data = JSON.parse(response.payload);
      expect(data.success).toBe(true);
      expect(data.token).toBeDefined();
      expect(data.token.isValid).toBe(true);
      expect(data.token.expiresAt).toBeDefined();
      expect(data.token.expiresInSeconds).toBeGreaterThan(0);
      expect(data.config).toBeDefined();
      expect(typeof data.config.jwtSecretConfigured).toBe('boolean');
    });

    it('should not expose secrets', async () => {
      const response = await app.inject({
        method: 'GET',
        url: '/api/auth/diagnostics',
        headers: {
          authorization: `Bearer ${validToken}`
        }
      });

      const data = JSON.parse(response.payload);
      // Should NOT contain actual secret values
      expect(JSON.stringify(data)).not.toContain(process.env.JWT_SECRET);
      expect(JSON.stringify(data)).not.toContain(process.env.SYNC_PASSWORD_HASH);
    });

    it('should return 401 without token', async () => {
      const response = await app.inject({
        method: 'GET',
        url: '/api/auth/diagnostics'
      });

      expect(response.statusCode).toBe(401);
    });
  });

  describe('POST /api/auth/change-password', () => {
    // Note: This test modifies .env, so skip in CI or use test isolation
    it.skip('should reject wrong current password', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/api/auth/change-password',
        headers: {
          authorization: `Bearer ${validToken}`
        },
        payload: {
          currentPassword: 'definitely-wrong',
          newPassword: 'newpass123'
        }
      });

      expect(response.statusCode).toBe(401);
      const data = JSON.parse(response.payload);
      expect(data.error).toBe('PASSWORD_INVALID');
    });

    it('should require authentication', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/api/auth/change-password',
        payload: {
          currentPassword: 'old',
          newPassword: 'new123'
        }
      });

      expect(response.statusCode).toBe(401);
    });

    it('should validate request body', async () => {
      const response = await app.inject({
        method: 'POST',
        url: '/api/auth/change-password',
        headers: {
          authorization: `Bearer ${validToken}`
        },
        payload: {
          currentPassword: '',
          newPassword: '123' // Too short
        }
      });

      // Zod validation should fail
      expect(response.statusCode).toBe(400);
    });
  });

  describe('Auth Middleware Error Codes', () => {
    it('should return TOKEN_MISSING for protected route without header', async () => {
      const response = await app.inject({
        method: 'GET',
        url: '/api/articulos'
      });

      expect(response.statusCode).toBe(401);
      const data = JSON.parse(response.payload);
      expect(data.error).toBe('TOKEN_MISSING');
    });

    it('should return TOKEN_INVALID for malformed token', async () => {
      const response = await app.inject({
        method: 'GET',
        url: '/api/articulos',
        headers: {
          authorization: 'Bearer not.a.valid.jwt'
        }
      });

      expect(response.statusCode).toBe(401);
      const data = JSON.parse(response.payload);
      expect(data.error).toBe('TOKEN_INVALID');
    });
  });
});
```
