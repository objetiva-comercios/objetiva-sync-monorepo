```typescript
import { useState } from 'react'
import { DashboardLayout } from '@objetiva/dashboard'
import OverviewPage from './pages/overview'
import MetricsPage from './pages/metrics'
import RecordsPage from './pages/records'

// Simple client-side routing based on path
function getPage(path: string) {
  switch (path) {
    case '/dashboard/metrics':
      return <MetricsPage />
    case '/dashboard/records':
      return <RecordsPage />
    default:
      return <OverviewPage />
  }
}

function App() {
  const [path] = useState(() => window.location.pathname)

  return (
    <DashboardLayout>
      {getPage(path)}
    </DashboardLayout>
  )
}

export default App
```

NOTE: Using simple path-based routing rather than react-router to minimize dependencies. Navigation uses regular anchor tags and full page loads (acceptable for dashboard).
